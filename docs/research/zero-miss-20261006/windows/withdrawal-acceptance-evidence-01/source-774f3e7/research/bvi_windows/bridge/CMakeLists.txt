cmake_minimum_required(VERSION 3.28)
project(PasBviCurrentPixelsOwnerOffline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(PAS_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
option(PAS_BRIDGE_ASAN "Sanitize the entire offline observer/owner/scheduler graph" OFF)
if(MSVC)
  add_compile_options(/W4 /permissive- /EHsc)
  if(PAS_BRIDGE_ASAN)
    if(NOT EXISTS "${PAS_ASAN_SUPPORT_ROOT}/sanitizer/common_interface_defs.h")
      message(FATAL_ERROR "Pinned sanitizer support missing")
    endif()
    file(COPY "${PAS_ASAN_SUPPORT_ROOT}/sanitizer" DESTINATION "${CMAKE_BINARY_DIR}/asan_support")
    file(GLOB ASAN_LIBS "${PAS_ASAN_SUPPORT_ROOT}/../lib/x64/*asan*.lib")
    file(COPY ${ASAN_LIBS} DESTINATION "${CMAKE_BINARY_DIR}/asan_support/lib")
    file(GLOB ASAN_DLLS "${PAS_ASAN_SUPPORT_ROOT}/../bin/Hostx64/x64/*asan*.dll")
    file(COPY ${ASAN_DLLS} DESTINATION "${CMAKE_BINARY_DIR}")
    include_directories("${CMAKE_BINARY_DIR}/asan_support")
    link_directories("${CMAKE_BINARY_DIR}/asan_support/lib")
    set(CMAKE_CXX_FLAGS_DEBUG "/MDd /Od /Ob0 /D_DEBUG")
    add_compile_options(/fsanitize=address)
    add_compile_definitions(_DISABLE_VECTOR_ANNOTATION _DISABLE_STRING_ANNOTATION _DISABLE_OPTIONAL_ANNOTATION)
    add_link_options(/INCREMENTAL:NO)
  endif()
endif()
# Recompile the real main pas_core source closure; no prebuilt unsanitized owner
# or Linux clock shim is substituted in any configuration.
file(STRINGS "${PAS_REPO}/CMakeLists.txt" CORE_LINE REGEX "^add_library\\(pas_core STATIC ")
if(NOT CORE_LINE)
  message(FATAL_ERROR "Formal core closure missing")
endif()
string(REGEX REPLACE "^add_library\\(pas_core STATIC (.*)\\)$" "\\1" CORE_FILES "${CORE_LINE}")
separate_arguments(CORE_FILES NATIVE_COMMAND "${CORE_FILES}")
list(TRANSFORM CORE_FILES PREPEND "${PAS_REPO}/")
add_library(formal_core STATIC ${CORE_FILES})
target_include_directories(formal_core PUBLIC "${PAS_REPO}/include")
target_link_libraries(formal_core PUBLIC nlohmann_json::nlohmann_json ZLIB::ZLIB
  bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32)
set(BVI "${PAS_REPO}/research/bvi_cold_v3")
add_library(exact_bvi_v3 STATIC "${BVI}/bvi.cpp" "${BVI}/contact_policy.cpp"
  "${BVI}/relation_policy.cpp" "${BVI}/constraint_policy.cpp")
target_include_directories(exact_bvi_v3 PUBLIC "${BVI}")
add_library(current_bridge STATIC bridge.cpp)
target_link_libraries(current_bridge PUBLIC formal_core exact_bvi_v3)
target_include_directories(current_bridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(bridge_tests bridge_tests.cpp)
target_link_libraries(bridge_tests PRIVATE current_bridge)
add_library(pixel_diagnostics STATIC pixel_diagnostics.cpp)
target_link_libraries(pixel_diagnostics PUBLIC formal_core exact_bvi_v3)
target_include_directories(pixel_diagnostics PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(pixel_chain pixel_chain.cpp)
target_link_libraries(pixel_chain PRIVATE current_bridge pixel_diagnostics)
add_executable(pixel_diagnostics_tests pixel_diagnostics_tests.cpp)
target_link_libraries(pixel_diagnostics_tests PRIVATE pixel_diagnostics)
add_executable(pixel_diagnostics_audit pixel_diagnostics_audit.cpp)
target_link_libraries(pixel_diagnostics_audit PRIVATE pixel_diagnostics)
enable_testing()
add_test(NAME complete_owner_prefix COMMAND bridge_tests "${CMAKE_BINARY_DIR}/prefix-results.json")
add_test(NAME bounded_post_dispatch_diagnostics COMMAND pixel_diagnostics_tests "${CMAKE_BINARY_DIR}/diagnostic-results.json")
