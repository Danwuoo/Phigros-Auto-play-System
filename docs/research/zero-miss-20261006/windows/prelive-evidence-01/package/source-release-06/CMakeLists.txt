set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(PAS_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
find_package(GTest CONFIG REQUIRED)
find_package(Protobuf CONFIG REQUIRED)
find_package(gRPC 1.81.1 EXACT CONFIG REQUIRED)
option(PAS_PRELIVE_ASAN "Sanitize the complete new closure" OFF)
if(MSVC)
 add_compile_options(/W4 /permissive- /EHsc)
 add_link_options(/STACK:8388608)
 add_link_options(/MAP)
 if(PAS_PRELIVE_ASAN)
  if(NOT EXISTS "${PAS_ASAN_SUPPORT_ROOT}/sanitizer/common_interface_defs.h")
   message(FATAL_ERROR "Pinned ASan support missing")
  endif()
  file(COPY "${PAS_ASAN_SUPPORT_ROOT}/sanitizer" DESTINATION "${CMAKE_BINARY_DIR}/asan_support")
  file(GLOB ASAN_LIBS "${PAS_ASAN_SUPPORT_ROOT}/../lib/x64/*asan*.lib")
  file(COPY ${ASAN_LIBS} DESTINATION "${CMAKE_BINARY_DIR}/asan_support/lib")
  file(GLOB ASAN_DLLS "${PAS_ASAN_SUPPORT_ROOT}/../bin/Hostx64/x64/*asan*.dll")
  file(COPY ${ASAN_DLLS} DESTINATION "${CMAKE_BINARY_DIR}")
  include_directories("${CMAKE_BINARY_DIR}/asan_support")
  link_directories("${CMAKE_BINARY_DIR}/asan_support/lib")
  set(CMAKE_CXX_FLAGS_DEBUG "/MDd /Od /Ob0 /D_DEBUG")
  add_compile_options(/fsanitize=address)
  add_compile_definitions(_DISABLE_VECTOR_ANNOTATION _DISABLE_STRING_ANNOTATION _DISABLE_OPTIONAL_ANNOTATION)
  add_link_options(/INCREMENTAL:NO)
 endif()
endif()
file(STRINGS "${PAS_REPO}/CMakeLists.txt" CORE_LINE REGEX "^add_library\\(pas_core STATIC ")
string(REGEX REPLACE "^add_library\\(pas_core STATIC (.*)\\)$" "\\1" CORE_FILES "${CORE_LINE}")

list(TRANSFORM CORE_FILES PREPEND "${PAS_REPO}/")

target_include_directories(formal_core PUBLIC "${PAS_REPO}/include")
target_link_libraries(formal_core PUBLIC nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32)

# Exact pure-contract prefix from the original runtime.cpp, before its first
# device-reading entry point. No test edits, replacements, oracle or endpoint.
file(READ "${PAS_REPO}/src/runtime.cpp" RUNTIME_TEXT)
string(FIND "${RUNTIME_TEXT}" "json game_preflight(" RUNTIME_DEVICE_START)
if(RUNTIME_DEVICE_START LESS 0)
 message(FATAL_ERROR "runtime pure-contract boundary missing")
endif()
string(SUBSTRING "${RUNTIME_TEXT}" 0 ${RUNTIME_DEVICE_START} RUNTIME_CONTRACT_TEXT)
file(WRITE "${CMAKE_BINARY_DIR}/runtime_contract.cpp" "${RUNTIME_CONTRACT_TEXT}\n} // namespace pas\n")
target_sources(formal_affected PRIVATE "${CMAKE_BINARY_DIR}/runtime_contract.cpp")
target_include_directories(formal_affected PRIVATE "${PAS_REPO}/out/win-r4-ninja-release-01/generated")
target_compile_definitions(formal_affected PRIVATE PAS_TEST_SOURCE_DIR="${PAS_REPO}")
target_link_libraries(formal_affected PRIVATE formal_core GTest::gtest_main protobuf::libprotobuf gRPC::grpc++)
