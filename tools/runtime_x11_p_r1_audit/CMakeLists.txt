cmake_minimum_required(VERSION 3.28)
project(R1ControllerAudit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
add_executable(r1_controller_audit main.cpp)
target_include_directories(r1_controller_audit PRIVATE "${R1_REPO}/out/x11-p-r1/B1/include")
target_link_libraries(r1_controller_audit PRIVATE "${R1_REPO}/out/x11-p-r1/build-B1/Release/pas_core.lib" nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32)
target_compile_options(r1_controller_audit PRIVATE /EHsc /utf-8 /W4)
