cmake_minimum_required(VERSION 3.28)
project(PasComparisonGraphControls LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT PAS_FROZEN_COMPARISON_REPO)
 message(FATAL_ERROR "Frozen comparison source repository required")
endif()
file(READ "${PAS_FROZEN_COMPARISON_REPO}/research/windows_cost_comparison/main.cpp" CONTROL_SOURCE)
string(REPLACE "int main(int argc,char** argv)" "int pas_comparison_control_entry(int argc,char** argv)" CONTROL_SOURCE "${CONTROL_SOURCE}")
string(REPLACE "\"../prelive_current/cost_analysis.cpp\"" "\"${PAS_FROZEN_COMPARISON_REPO}/research/prelive_current/cost_analysis.cpp\"" CONTROL_SOURCE "${CONTROL_SOURCE}")
string(REPLACE "\"../windows_qualification/verifier.hpp\"" "\"${PAS_FROZEN_COMPARISON_REPO}/research/windows_qualification/verifier.hpp\"" CONTROL_SOURCE "${CONTROL_SOURCE}")
string(REPLACE "\"metrics.hpp\"" "\"${PAS_FROZEN_COMPARISON_REPO}/research/windows_cost_comparison/metrics.hpp\"" CONTROL_SOURCE "${CONTROL_SOURCE}")
file(WRITE "${CMAKE_BINARY_DIR}/comparison_control.inc" "${CONTROL_SOURCE}")
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(comparison_graph_controls main.cpp)
target_include_directories(comparison_graph_controls PRIVATE "${CMAKE_BINARY_DIR}")
target_link_libraries(comparison_graph_controls PRIVATE nlohmann_json::nlohmann_json bcrypt)
if(MSVC)
 target_compile_options(comparison_graph_controls PRIVATE /W4 /permissive- /EHsc)
 target_link_options(comparison_graph_controls PRIVATE /STACK:8388608 /MAP)
endif()
