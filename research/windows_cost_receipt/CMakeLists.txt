cmake_minimum_required(VERSION 3.28)
project(PasWindowsCostReceipt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(pas_cost_receipt_verify main.cpp)
target_link_libraries(pas_cost_receipt_verify PRIVATE nlohmann_json::nlohmann_json)
if(MSVC)
 target_compile_options(pas_cost_receipt_verify PRIVATE /W4 /permissive- /EHsc)
 target_link_options(pas_cost_receipt_verify PRIVATE /STACK:8388608)
endif()
