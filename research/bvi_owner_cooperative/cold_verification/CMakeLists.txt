cmake_minimum_required(VERSION 3.28)
project(PasUniformRgbFnvColdVerification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT PAS_WORKSPACE OR NOT PAS_ORIGINAL_BUILD)
 message(FATAL_ERROR "Explicit workspace and preserved donor build required")
endif()
find_package(nlohmann_json CONFIG REQUIRED)
add_library(exact_bvi_v3 STATIC IMPORTED GLOBAL)
set_target_properties(exact_bvi_v3 PROPERTIES IMPORTED_LOCATION "${PAS_ORIGINAL_BUILD}/cold/exact_bvi_v3.lib")
add_executable(uniform_rgb_fnv_controls main.cpp)
target_include_directories(uniform_rgb_fnv_controls PRIVATE "${PAS_WORKSPACE}/research/bvi_cold_v3")
target_link_libraries(uniform_rgb_fnv_controls PRIVATE exact_bvi_v3 nlohmann_json::nlohmann_json)
target_compile_options(uniform_rgb_fnv_controls PRIVATE /W4 /permissive- /EHsc)
target_link_options(uniform_rgb_fnv_controls PRIVATE /INCREMENTAL:NO /STACK:8388608)
