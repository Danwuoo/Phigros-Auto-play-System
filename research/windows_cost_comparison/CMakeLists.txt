cmake_minimum_required(VERSION 3.28)
project(PasWindowsCostComparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(pas_cost_comparison main.cpp)
target_link_libraries(pas_cost_comparison PRIVATE nlohmann_json::nlohmann_json bcrypt)
add_executable(tap_identity_diagnose tap_identity_diagnose.cpp)
target_link_libraries(tap_identity_diagnose PRIVATE nlohmann_json::nlohmann_json bcrypt)
add_executable(comparison_metrics_tests metrics_tests.cpp)
target_link_libraries(comparison_metrics_tests PRIVATE nlohmann_json::nlohmann_json)
if(MSVC)
 target_compile_options(comparison_metrics_tests PRIVATE /W4 /permissive- /EHsc)
 target_link_options(comparison_metrics_tests PRIVATE /STACK:8388608)
 target_compile_options(pas_cost_comparison PRIVATE /W4 /permissive- /EHsc)
 target_link_options(pas_cost_comparison PRIVATE /STACK:8388608)
endif()
