cmake_minimum_required(VERSION 3.28)
project(PasFunctionalEnvelope LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(pas_functional_envelope WIN32 runner.cpp)
target_link_libraries(pas_functional_envelope PRIVATE nlohmann_json::nlohmann_json)
target_compile_options(pas_functional_envelope PRIVATE /W4 /permissive- /EHsc)
target_link_options(pas_functional_envelope PRIVATE /STACK:8388608)
