cmake_minimum_required(VERSION 3.28)
project(PasWindowsQualification LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(pas_qualification_verify main.cpp)
target_link_libraries(pas_qualification_verify PRIVATE nlohmann_json::nlohmann_json)
add_executable(pas_qualification_tests tests.cpp)
add_executable(pas_cost_corpus corpus.cpp)
target_link_libraries(pas_qualification_tests PRIVATE nlohmann_json::nlohmann_json)
target_link_libraries(pas_cost_corpus PRIVATE nlohmann_json::nlohmann_json)
if(MSVC)
  foreach(target pas_qualification_verify pas_qualification_tests pas_cost_corpus)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
    target_link_options(${target} PRIVATE /STACK:8388608)
  endforeach()
endif()
