cmake_minimum_required(VERSION 3.28)
project(PasJournalCostCandidate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Rebuild the complete formal core: its public headers and archive/journal
# implementation changed. No old compiled core or current library is reused.
add_subdirectory(../prelive_current prelive)
set(PAS_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_executable(producer_wait_tests "${PAS_REPO}/research/prelive_current/tests/producer_wait_tests.cpp")
add_executable(tap_reset_workload_tests "${PAS_REPO}/research/prelive_current/tests/tap_reset_workload_tests.cpp")
target_link_libraries(producer_wait_tests PRIVATE current_rails)
target_link_libraries(tap_reset_workload_tests PRIVATE current_rails)
add_executable(journal_encoding_tests "${PAS_REPO}/tests/journal_encoding_tests.cpp")
target_link_libraries(journal_encoding_tests PRIVATE formal_core GTest::gtest_main)
add_executable(original_session_tests "${PAS_REPO}/tests/game_session_tests.cpp")
target_link_libraries(original_session_tests PRIVATE formal_core GTest::gtest_main)
