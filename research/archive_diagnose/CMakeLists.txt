cmake_minimum_required(VERSION 3.28)
project(PasArchiveDiagnosis LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
add_executable(archive_diagnose main.cpp)
target_include_directories(archive_diagnose PRIVATE ../../include)
target_link_libraries(archive_diagnose PRIVATE nlohmann_json::nlohmann_json ZLIB::ZLIB)
if(MSVC)
 target_compile_options(archive_diagnose PRIVATE /W4 /permissive- /EHsc)
endif()
