cmake_minimum_required(VERSION 3.28)
project(PasOfflineResearch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT MSVC)
  message(FATAL_ERROR "This research probe uses the production MSVC configuration")
endif()
set(PAS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${PAS_ROOT}/src/game_motion.cpp" "${PAS_ROOT}/include/pas/game.hpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/line_identity_probe.cpp")
set(PAS_DEPENDENCY_INCLUDE "${PAS_ROOT}/out/vcpkg_installed/x64-windows/include" CACHE PATH "Installed dependency headers")
add_executable(pas_line_identity_probe line_identity_probe.cpp "${PAS_ROOT}/src/game_motion.cpp")
target_include_directories(pas_line_identity_probe PRIVATE "${PAS_ROOT}/include" "${PAS_DEPENDENCY_INCLUDE}")
target_compile_options(pas_line_identity_probe PRIVATE /W4 /permissive- /EHsc)
file(SHA256 "${PAS_ROOT}/src/game_motion.cpp" PAS_MOTION_SHA256)
file(SHA256 "${PAS_ROOT}/include/pas/game.hpp" PAS_GAME_HEADER_SHA256)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/line_identity_probe.cpp" PAS_PROBE_SHA256)
configure_file(provenance.hpp.in provenance.hpp @ONLY)
target_include_directories(pas_line_identity_probe PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_executable(pas_line_policy_bench line_policy_bench.cpp "${PAS_ROOT}/src/game_motion.cpp")
target_include_directories(pas_line_policy_bench PRIVATE "${PAS_ROOT}/include" "${PAS_DEPENDENCY_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(pas_line_policy_bench PRIVATE /W4 /permissive- /EHsc)
set(PAS_CORE_LIBRARY "${PAS_ROOT}/out/release-v145/Release/pas_core.lib" CACHE FILEPATH "Current development pas_core library for offline pixel replay")
add_executable(pas_pixel_clip_probe pixel_clip_probe.cpp)
target_include_directories(pas_pixel_clip_probe PRIVATE "${PAS_ROOT}/include" "${PAS_DEPENDENCY_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(pas_pixel_clip_probe PRIVATE "${PAS_CORE_LIBRARY}" bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32)
target_compile_options(pas_pixel_clip_probe PRIVATE /W4 /permissive- /EHsc)
add_executable(pas_line_assignment_compare line_assignment_compare.cpp "${PAS_ROOT}/src/game_motion.cpp")
target_include_directories(pas_line_assignment_compare PRIVATE "${PAS_ROOT}/include" "${PAS_DEPENDENCY_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(pas_line_assignment_compare PRIVATE /W4 /permissive- /EHsc)
