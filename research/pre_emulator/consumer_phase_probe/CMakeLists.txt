cmake_minimum_required(VERSION 3.28)
project(PasConsumerPhaseProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT PAS_WORKSPACE OR NOT PAS_ORIGINAL_BUILD)
 message(FATAL_ERROR "Explicit workspace and preserved base Release build required")
endif()
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
foreach(lib current_rails current_front own_bridge exact_bvi_v3)
 add_library(${lib} STATIC IMPORTED GLOBAL)
 set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${PAS_ORIGINAL_BUILD}/cold/${lib}.lib")
endforeach()
add_library(pas_core STATIC IMPORTED GLOBAL)
set_target_properties(pas_core PROPERTIES IMPORTED_LOCATION "${PAS_ORIGINAL_BUILD}/production/pas_core.lib")
# Profile and OwnerCandidateStep are inline and instantiated only in this new
# chain TU. None of the five imported libraries contains their changed layout.
# pas_emulator (manual_session) and any historical chain object are excluded.
add_executable(current_chain_consumer_phase_probe "${PAS_WORKSPACE}/research/prelive_current/chain.cpp")
target_compile_definitions(current_chain_consumer_phase_probe PRIVATE PAS_DIAGNOSTIC_PHASE_PROFILE=1)
target_include_directories(current_chain_consumer_phase_probe PRIVATE "${PAS_WORKSPACE}/include" "${PAS_WORKSPACE}/research/prelive_current" "${PAS_WORKSPACE}/research/bvi_cold_v3" "${PAS_WORKSPACE}/research/hold_front_current")
target_link_libraries(current_chain_consumer_phase_probe PRIVATE current_rails current_front own_bridge exact_bvi_v3 pas_core nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32 psapi)
add_executable(consumer_phase_summary "${PAS_WORKSPACE}/research/prelive_current/diagnostic_phase_summary.cpp")
target_include_directories(consumer_phase_summary PRIVATE "${PAS_WORKSPACE}/research/prelive_current")
target_link_libraries(consumer_phase_summary PRIVATE nlohmann_json::nlohmann_json)
foreach(target current_chain_consumer_phase_probe consumer_phase_summary)
 target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
 target_link_options(${target} PRIVATE /INCREMENTAL:NO /STACK:8388608)
endforeach()
add_custom_target(consumer_phase_probe DEPENDS current_chain_consumer_phase_probe consumer_phase_summary)
file(COPY "${PAS_ORIGINAL_BUILD}/cold/z.dll" DESTINATION "${CMAKE_BINARY_DIR}")
