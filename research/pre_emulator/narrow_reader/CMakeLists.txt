cmake_minimum_required(VERSION 3.28)
project(PasNarrowPreflightReader LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT PAS_WORKSPACE)
 message(FATAL_ERROR "Explicit current main workspace required")
endif()
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(qualification_reader "${PAS_WORKSPACE}/research/pre_emulator/qualification_reader/main.cpp")
add_executable(qualification_reader_controls "${PAS_WORKSPACE}/research/pre_emulator/qualification_reader/controls.cpp")
foreach(target qualification_reader qualification_reader_controls)
 target_include_directories(${target} PRIVATE "${PAS_WORKSPACE}")
 target_link_libraries(${target} PRIVATE nlohmann_json::nlohmann_json bcrypt)
 target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
 target_link_options(${target} PRIVATE /INCREMENTAL:NO /STACK:8388608)
endforeach()
