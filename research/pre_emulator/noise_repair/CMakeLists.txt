cmake_minimum_required(VERSION 3.28)
project(PasNarrowNoiseRepair LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT PAS_WORKSPACE OR NOT PAS_ORIGINAL_BUILD)
 message(FATAL_ERROR "Explicit workspace and preserved base Release build required")
endif()
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
set(PAS_ORIGINAL_COLD "${PAS_ORIGINAL_BUILD}/cold")
foreach(lib current_rails current_front exact_bvi_v3)
 add_library(${lib} STATIC IMPORTED GLOBAL)
 set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${PAS_ORIGINAL_COLD}/${lib}.lib")
endforeach()
# CPU repair compiles the complete formal core and cooperative bridge afresh.
# Original libraries remain preserved prerequisites, never effective inputs for
# these two roles. This does not repeat the original 75/prod9 test campaigns.
set(PAS_CORE_SOURCES core config analysis adb journal touch_evidence preview game
 game_clip_replay game_cold_pipeline game_cold_corpus game_motion game_tracking
 game_tracking_methods game_tracking_analysis game_tracking_shadow game_dataset
 game_dataset_tools game_session session_archive session_pixel_clips session_recording)
set(PAS_CORE_FILES)
foreach(source IN LISTS PAS_CORE_SOURCES)
 list(APPEND PAS_CORE_FILES "${PAS_WORKSPACE}/src/${source}.cpp")
endforeach()
add_library(pas_core STATIC ${PAS_CORE_FILES})
target_include_directories(pas_core PUBLIC "${PAS_WORKSPACE}/include")
target_link_libraries(pas_core PUBLIC nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32)
add_library(own_bridge STATIC "${PAS_WORKSPACE}/research/prelive_current/own_bridge.cpp" "${PAS_WORKSPACE}/research/bvi_owner_cooperative/extract.cpp")
target_include_directories(own_bridge PUBLIC "${PAS_WORKSPACE}/research/prelive_current" "${PAS_WORKSPACE}/research/bvi_cold_v3")
target_link_libraries(own_bridge PUBLIC pas_core exact_bvi_v3)
add_executable(current_chain "${PAS_WORKSPACE}/research/prelive_current/chain.cpp")
target_include_directories(current_chain PRIVATE "${PAS_WORKSPACE}/include" "${PAS_WORKSPACE}/research/prelive_current" "${PAS_WORKSPACE}/research/bvi_cold_v3" "${PAS_WORKSPACE}/research/hold_front_current")
target_link_libraries(current_chain PRIVATE current_rails current_front own_bridge exact_bvi_v3 pas_core nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32 psapi)
add_executable(consumer_priority_controls "${PAS_WORKSPACE}/research/prelive_current/tests/consumer_priority_tests.cpp")
target_include_directories(consumer_priority_controls PRIVATE "${PAS_WORKSPACE}/include" "${PAS_WORKSPACE}/research/prelive_current")
target_link_libraries(consumer_priority_controls PRIVATE nlohmann_json::nlohmann_json)
add_executable(worker_affinity_controls "${PAS_WORKSPACE}/research/prelive_current/tests/worker_affinity_tests.cpp")
target_include_directories(worker_affinity_controls PRIVATE "${PAS_WORKSPACE}/include" "${PAS_WORKSPACE}/research/prelive_current")
target_link_libraries(worker_affinity_controls PRIVATE nlohmann_json::nlohmann_json)
add_executable(producer_wait_tests "${PAS_WORKSPACE}/research/prelive_current/tests/producer_wait_tests.cpp")
target_include_directories(producer_wait_tests PRIVATE "${PAS_WORKSPACE}/include" "${PAS_WORKSPACE}/research/prelive_current")
# Only HostClock's QPC definitions are pulled from the fresh core.
target_link_libraries(producer_wait_tests PRIVATE pas_core nlohmann_json::nlohmann_json)
foreach(target owner_checkpoint_tests owner_expiry_tests)
 add_executable(${target} "${PAS_WORKSPACE}/research/prelive_current/tests/${target}.cpp")
 target_include_directories(${target} PRIVATE "${PAS_WORKSPACE}/include" "${PAS_WORKSPACE}/research/prelive_current" "${PAS_WORKSPACE}/research/bvi_cold_v3" "${PAS_WORKSPACE}/research/hold_front_current")
 target_link_libraries(${target} PRIVATE current_rails current_front own_bridge exact_bvi_v3 pas_core nlohmann_json::nlohmann_json ZLIB::ZLIB)
endforeach()
add_executable(component_mask_tests "${PAS_WORKSPACE}/research/prelive_current/tests/component_mask_tests.cpp")
target_include_directories(component_mask_tests PRIVATE "${PAS_WORKSPACE}/include" "${PAS_WORKSPACE}/research/prelive_current")
target_link_libraries(component_mask_tests PRIVATE pas_core nlohmann_json::nlohmann_json)
add_executable(uniform_fnv_tests "${PAS_WORKSPACE}/research/bvi_owner_cooperative/cold_verification/main.cpp")
target_include_directories(uniform_fnv_tests PRIVATE "${PAS_WORKSPACE}/include" "${PAS_WORKSPACE}/research/bvi_cold_v3")
target_link_libraries(uniform_fnv_tests PRIVATE exact_bvi_v3 nlohmann_json::nlohmann_json)
foreach(target pas_core own_bridge current_chain consumer_priority_controls worker_affinity_controls producer_wait_tests owner_checkpoint_tests owner_expiry_tests component_mask_tests uniform_fnv_tests)
 target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
 if(NOT target STREQUAL "pas_core" AND NOT target STREQUAL "own_bridge")
  target_link_options(${target} PRIVATE /INCREMENTAL:NO /STACK:8388608)
 endif()
endforeach()
file(COPY "${PAS_ORIGINAL_COLD}/z.dll" DESTINATION "${CMAKE_BINARY_DIR}")
