cmake_minimum_required(VERSION 3.28)
project(PasNarrowNoiseRepair LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT PAS_WORKSPACE OR NOT PAS_ORIGINAL_BUILD)
 message(FATAL_ERROR "Explicit workspace and preserved base Release build required")
endif()
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
set(PAS_ORIGINAL_COLD "${PAS_ORIGINAL_BUILD}/cold")
foreach(lib current_rails current_front own_bridge exact_bvi_v3)
 add_library(${lib} STATIC IMPORTED GLOBAL)
 set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${PAS_ORIGINAL_COLD}/${lib}.lib")
endforeach()
add_library(pas_core STATIC IMPORTED GLOBAL)
set_target_properties(pas_core PROPERTIES IMPORTED_LOCATION "${PAS_ORIGINAL_BUILD}/production/pas_core.lib")
add_executable(current_chain "${PAS_WORKSPACE}/research/prelive_current/chain.cpp")
target_include_directories(current_chain PRIVATE "${PAS_WORKSPACE}/include" "${PAS_WORKSPACE}/research/prelive_current" "${PAS_WORKSPACE}/research/bvi_cold_v3" "${PAS_WORKSPACE}/research/hold_front_current")
target_link_libraries(current_chain PRIVATE current_rails current_front own_bridge exact_bvi_v3 pas_core nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32 psapi)
add_executable(consumer_priority_controls "${PAS_WORKSPACE}/research/prelive_current/tests/consumer_priority_tests.cpp")
target_include_directories(consumer_priority_controls PRIVATE "${PAS_WORKSPACE}/include" "${PAS_WORKSPACE}/research/prelive_current")
target_link_libraries(consumer_priority_controls PRIVATE nlohmann_json::nlohmann_json)
add_executable(worker_affinity_controls "${PAS_WORKSPACE}/research/prelive_current/tests/worker_affinity_tests.cpp")
target_include_directories(worker_affinity_controls PRIVATE "${PAS_WORKSPACE}/include" "${PAS_WORKSPACE}/research/prelive_current")
target_link_libraries(worker_affinity_controls PRIVATE nlohmann_json::nlohmann_json)
foreach(target current_chain consumer_priority_controls worker_affinity_controls)
 target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
 target_link_options(${target} PRIVATE /INCREMENTAL:NO /STACK:8388608)
endforeach()
file(COPY "${PAS_ORIGINAL_COLD}/z.dll" DESTINATION "${CMAKE_BINARY_DIR}")
