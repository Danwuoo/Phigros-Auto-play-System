cmake_minimum_required(VERSION 3.28)
project(PasJournalCandidateMaintenance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT PAS_JOURNAL_SOURCE_REPO)
 message(FATAL_ERROR "Frozen journal candidate source required")
endif()
find_package(GTest CONFIG REQUIRED)
option(PAS_PRELIVE_ASAN "Instrument all candidate and test translation units" OFF)
if(MSVC)
 add_compile_options(/W4 /permissive- /EHsc)
 add_link_options(/STACK:8388608 /MAP)
 if(PAS_PRELIVE_ASAN)
  set(CMAKE_CXX_FLAGS_DEBUG "/MDd /Od /Ob0 /D_DEBUG")
  add_compile_options(/fsanitize=address)
  add_compile_definitions(_DISABLE_VECTOR_ANNOTATION _DISABLE_STRING_ANNOTATION _DISABLE_OPTIONAL_ANNOTATION)
  include_directories("${CMAKE_BINARY_DIR}/asan_support")
  link_directories("${CMAKE_BINARY_DIR}/asan_support/lib")
  add_link_options(/INCREMENTAL:NO)
 endif()
endif()
add_subdirectory("${PAS_JOURNAL_SOURCE_REPO}/research/windows_journal_cost" candidate)
# Match the unchanged original repository's GTest source-data definition.
# This repairs the standalone build, without rewriting a test or its oracle.
target_compile_definitions(original_session_tests PRIVATE PAS_TEST_SOURCE_DIR="${PAS_JOURNAL_SOURCE_REPO}")
