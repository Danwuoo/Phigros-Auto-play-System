cmake_minimum_required(VERSION 3.28)
project(PasColdSchedulerDiagnostic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(VCPKG_APPLOCAL_DEPS OFF CACHE BOOL "Standalone Windows diagnostic tools" FORCE)
foreach(name PAS_SOURCE_WORKSPACE PAS_DEPENDENCY_INCLUDE PAS_TOOL_SOURCE_COMMIT
        PAS_CONTROLLER_SOURCE_SHA256 PAS_CONTROLLER_TEST_SOURCE_SHA256 PAS_READER_SOURCE_SHA256)
 if(NOT DEFINED ${name} OR "${${name}}" STREQUAL "")
  message(FATAL_ERROR "Required diagnostic build input: ${name}")
 endif()
endforeach()
string(LENGTH "${PAS_TOOL_SOURCE_COMMIT}" commit_length)
if(NOT commit_length EQUAL 40 OR NOT PAS_TOOL_SOURCE_COMMIT MATCHES "^[0-9a-f]+$")
 message(FATAL_ERROR "Exact frozen tool source commit required")
endif()
function(require_source_sha path expected)
 if(NOT EXISTS "${path}")
  message(FATAL_ERROR "Missing diagnostic source: ${path}")
 endif()
 file(SHA256 "${path}" actual)
 if(NOT actual STREQUAL expected)
  message(FATAL_ERROR "Diagnostic source SHA mismatch: ${path}")
 endif()
endfunction()
require_source_sha("${PAS_SOURCE_WORKSPACE}/tools/cold_scheduler_trace_controller.cpp" "${PAS_CONTROLLER_SOURCE_SHA256}")
require_source_sha("${PAS_SOURCE_WORKSPACE}/tests/cold_scheduler_trace_controller_tests.cpp" "${PAS_CONTROLLER_TEST_SOURCE_SHA256}")
require_source_sha("${PAS_SOURCE_WORKSPACE}/tools/cold_scheduler_trace_reader.cpp" "${PAS_READER_SOURCE_SHA256}")
if(NOT EXISTS "${PAS_DEPENDENCY_INCLUDE}/nlohmann/json.hpp")
 message(FATAL_ERROR "Existing installed nlohmann headers required")
endif()
add_executable(cold_scheduler_trace_controller "${PAS_SOURCE_WORKSPACE}/tools/cold_scheduler_trace_controller.cpp")
add_executable(cold_scheduler_trace_controller_tests "${PAS_SOURCE_WORKSPACE}/tests/cold_scheduler_trace_controller_tests.cpp")
add_executable(cold_scheduler_trace_reader "${PAS_SOURCE_WORKSPACE}/tools/cold_scheduler_trace_reader.cpp")
foreach(target cold_scheduler_trace_controller cold_scheduler_trace_controller_tests cold_scheduler_trace_reader)
 target_include_directories(${target} PRIVATE "${PAS_SOURCE_WORKSPACE}" "${PAS_SOURCE_WORKSPACE}/include" "${PAS_DEPENDENCY_INCLUDE}")
 target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE
  PAS_COLD_SCHEDULER_TOOL_SOURCE_COMMIT="${PAS_TOOL_SOURCE_COMMIT}" PAS_COLD_SCHEDULER_TOOL_SOURCE_DIRTY=0)
 target_link_libraries(${target} PRIVATE advapi32 bcrypt ole32 tdh)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
 endif()
endforeach()
