cmake_minimum_required(VERSION 3.28)
project(PasColdSchedulerReaderDelta LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(VCPKG_APPLOCAL_DEPS OFF CACHE BOOL "Standalone Windows SDK reader" FORCE)
foreach(name PAS_SOURCE_WORKSPACE PAS_TOOL_SOURCE_COMMIT PAS_READER_SOURCE_SHA256)
 if(NOT DEFINED ${name} OR "${${name}}" STREQUAL "")
  message(FATAL_ERROR "Required reader delta input: ${name}")
 endif()
endforeach()
string(LENGTH "${PAS_TOOL_SOURCE_COMMIT}" commit_length)
if(NOT commit_length EQUAL 40 OR NOT PAS_TOOL_SOURCE_COMMIT MATCHES "^[0-9a-f]+$")
 message(FATAL_ERROR "Exact frozen reader source commit required")
endif()
set(reader_source "${PAS_SOURCE_WORKSPACE}/tools/cold_scheduler_trace_reader.cpp")
if(NOT EXISTS "${reader_source}")
 message(FATAL_ERROR "Missing frozen reader source")
endif()
file(SHA256 "${reader_source}" reader_sha256)
if(NOT reader_sha256 STREQUAL PAS_READER_SOURCE_SHA256)
 message(FATAL_ERROR "Frozen reader source SHA mismatch")
endif()
add_executable(cold_scheduler_trace_reader "${reader_source}")
target_compile_definitions(cold_scheduler_trace_reader PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE
 PAS_COLD_SCHEDULER_TOOL_SOURCE_COMMIT="${PAS_TOOL_SOURCE_COMMIT}" PAS_COLD_SCHEDULER_TOOL_SOURCE_DIRTY=0)
target_link_libraries(cold_scheduler_trace_reader PRIVATE advapi32 tdh)
if(MSVC)
 target_compile_options(cold_scheduler_trace_reader PRIVATE /W4 /permissive- /EHsc)
endif()
