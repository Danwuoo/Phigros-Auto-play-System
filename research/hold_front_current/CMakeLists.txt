cmake_minimum_required(VERSION 3.28)
project(PasCurrentHoldFront LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(PAS_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
option(PAS_FRONT_ASAN "MSVC ASan new targets plus matching preverified ASan closure" OFF)
if(MSVC)
  add_compile_options(/W4 /permissive- /EHsc)
  add_link_options(/STACK:8388608)
  if(PAS_FRONT_ASAN)
    file(COPY "${PAS_ASAN_SUPPORT_ROOT}/sanitizer" DESTINATION "${CMAKE_BINARY_DIR}/asan_support")
    file(GLOB ASAN_LIBS "${PAS_ASAN_SUPPORT_ROOT}/../lib/x64/*asan*.lib")
    file(COPY ${ASAN_LIBS} DESTINATION "${CMAKE_BINARY_DIR}/asan_support/lib")
    file(GLOB ASAN_DLLS "${PAS_ASAN_SUPPORT_ROOT}/../bin/Hostx64/x64/*asan*.dll")
    file(COPY ${ASAN_DLLS} DESTINATION "${CMAKE_BINARY_DIR}")
    include_directories("${CMAKE_BINARY_DIR}/asan_support")
    link_directories("${CMAKE_BINARY_DIR}/asan_support/lib")
    set(CMAKE_CXX_FLAGS_DEBUG "/MDd /Od /Ob0 /D_DEBUG")
    add_compile_options(/fsanitize=address)
    add_compile_definitions(_DISABLE_VECTOR_ANNOTATION _DISABLE_STRING_ANNOTATION _DISABLE_OPTIONAL_ANNOTATION)
    add_link_options(/INCREMENTAL:NO)
  endif()
endif()
if(NOT PAS_CLOSURE_MANIFEST)
  message(FATAL_ERROR "Supply the checked same-configuration source/binary closure manifest")
endif()
file(READ "${PAS_CLOSURE_MANIFEST}" CLOSURE)
string(JSON EXPECT_CONFIG GET "${CLOSURE}" configuration)
if(NOT EXPECT_CONFIG STREQUAL CMAKE_BUILD_TYPE)
  message(FATAL_ERROR "Closure configuration mismatch")
endif()
string(JSON EXPECT_ASAN GET "${CLOSURE}" asan)
if(NOT EXPECT_ASAN STREQUAL PAS_FRONT_ASAN)
  message(FATAL_ERROR "Closure ASan mismatch")
endif()
foreach(name formal_core exact_bvi_v3 current_bridge pixel_diagnostics)
  string(JSON PATH GET "${CLOSURE}" libraries ${name} path)
  string(JSON EXPECT_SHA GET "${CLOSURE}" libraries ${name} sha256)
  file(SHA256 "${PATH}" ACTUAL_SHA)
  if(NOT ACTUAL_SHA STREQUAL EXPECT_SHA)
    message(FATAL_ERROR "Linked closure SHA mismatch: ${name}")
  endif()
  add_library(${name} STATIC IMPORTED)
  set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${PATH}")
endforeach()
set(BRIDGE "${PAS_REPO}/research/bvi_windows/bridge")
target_include_directories(formal_core INTERFACE "${PAS_REPO}/include")
target_link_libraries(formal_core INTERFACE nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32)
target_include_directories(exact_bvi_v3 INTERFACE "${PAS_REPO}/research/bvi_cold_v3")
target_include_directories(current_bridge INTERFACE "${BRIDGE}")
target_link_libraries(current_bridge INTERFACE formal_core exact_bvi_v3)
target_link_libraries(pixel_diagnostics INTERFACE current_bridge)
add_library(current_front STATIC front.cpp)
target_include_directories(current_front PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(current_front PUBLIC current_bridge)
add_executable(front_tests front_tests.cpp)
target_link_libraries(front_tests PRIVATE current_front)
add_executable(prefix_preservation "${BRIDGE}/bridge_tests.cpp")
target_link_libraries(prefix_preservation PRIVATE current_bridge)
add_executable(pixel_chain front_chain.cpp)
target_link_libraries(pixel_chain PRIVATE current_front pixel_diagnostics)
add_executable(front_audit front_audit.cpp)
target_link_libraries(front_audit PRIVATE pixel_diagnostics)
