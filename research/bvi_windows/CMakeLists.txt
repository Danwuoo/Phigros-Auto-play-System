# COPYONLY prevents a quoted bvi.hpp include in a historical fixture from
# resolving to its adjacent v2 header. Test and oracle bytes stay unchanged.
function(bvi_fixture name source)
  configure_file("${source}" "${CMAKE_BINARY_DIR}/frozen-fixtures/${name}.cpp" COPYONLY)
  add_executable(${name} "${CMAKE_BINARY_DIR}/frozen-fixtures/${name}.cpp")
  target_link_libraries(${name} PRIVATE bvi_v3)
  target_compile_definitions(${name} PRIVATE BVI_HEADER="${BVI}/bvi.hpp")
endfunction()
