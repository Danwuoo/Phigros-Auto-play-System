cmake_minimum_required(VERSION 3.28)
project(PasSessionBaselineControl LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT PAS_BASELINE_SOURCE OR NOT PAS_BASELINE_CORE)
 message(FATAL_ERROR "Exact frozen source and production core required")
endif()
find_package(GTest CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
add_library(frozen_core STATIC IMPORTED GLOBAL)
set_target_properties(frozen_core PROPERTIES IMPORTED_LOCATION "${PAS_BASELINE_CORE}")
add_executable(original_session_baseline "${PAS_BASELINE_SOURCE}/tests/game_session_tests.cpp")
target_include_directories(original_session_baseline PRIVATE "${PAS_BASELINE_SOURCE}/include")
target_compile_definitions(original_session_baseline PRIVATE PAS_TEST_SOURCE_DIR="${PAS_BASELINE_SOURCE}")
target_link_libraries(original_session_baseline PRIVATE frozen_core GTest::gtest_main nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32)
add_executable(session_trace_compare main.cpp)
target_link_libraries(session_trace_compare PRIVATE nlohmann_json::nlohmann_json)
if(MSVC)
 add_compile_options(/W4 /permissive- /EHsc)
 target_link_options(original_session_baseline PRIVATE /STACK:8388608)
 target_link_options(session_trace_compare PRIVATE /STACK:8388608)
endif()
