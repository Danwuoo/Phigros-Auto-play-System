cmake_minimum_required(VERSION 3.28)
project(X10dOwnership LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(TO_CMAKE_PATH "${O_REPO}" O_REPO)
set(root ${O_REPO}/out/x10d-o/reference)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
find_package(GTest CONFIG REQUIRED)
set(parts core game game_tracking game_motion game_session analysis adb config game_tracking_methods touch_evidence session_archive session_pixel_clips journal game_tracking_shadow)
set(sources)
foreach(p IN LISTS parts)
 list(APPEND sources ${root}/src/${p}.cpp)
endforeach()
add_library(o_core STATIC ${sources})
target_include_directories(o_core PUBLIC ${root}/include ${O_REPO}/apps/frame_review)
target_link_libraries(o_core PUBLIC nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 user32)
target_compile_options(o_core PRIVATE /W4 /permissive- /EHsc /utf-8)
add_executable(original_tests ${O_REPO}/tests/contact_replay_tests.cpp ${O_REPO}/apps/frame_review/x10c_contact_replay.cpp
 ${root}/tests/game_tests_offline.cpp ${root}/tests/game_tracking_tests.cpp ${root}/tests/game_session_tests.cpp ${root}/tests/core_tests.cpp)
target_include_directories(original_tests PRIVATE ${root}/tests)
target_compile_definitions(original_tests PRIVATE PAS_TEST_SOURCE_DIR="${O_REPO}")
target_compile_options(original_tests PRIVATE /EHsc /utf-8 /bigobj)
target_link_libraries(original_tests PRIVATE o_core GTest::gtest_main)
add_executable(reference_contract contract_tests.cpp reference.cpp)
target_link_libraries(reference_contract PRIVATE o_core GTest::gtest_main)
target_compile_options(reference_contract PRIVATE /W4 /EHsc /utf-8)
if(O_CANDIDATE)
 add_executable(candidate_contract contract_tests.cpp candidate.cpp)
 target_link_libraries(candidate_contract PRIVATE o_core GTest::gtest_main)
 target_compile_options(candidate_contract PRIVATE /W4 /EHsc /utf-8)
 add_executable(owner_controls owner_controls.cpp)
 target_link_libraries(owner_controls PRIVATE o_core GTest::gtest_main)
 target_compile_options(owner_controls PRIVATE /W4 /EHsc /utf-8)
 add_executable(packet_audit packet_audit.cpp candidate.cpp)
 target_link_libraries(packet_audit PRIVATE o_core)
 target_compile_options(packet_audit PRIVATE /W4 /EHsc /utf-8 /bigobj)
endif()
if(O_ASAN_SUPPORT)
 file(COPY ${O_ASAN_SUPPORT}/sanitizer DESTINATION ${CMAKE_BINARY_DIR}/asan_support)
 file(GLOB libs "${O_ASAN_SUPPORT}/../lib/x64/*asan*.lib")
 file(COPY ${libs} DESTINATION ${CMAKE_BINARY_DIR}/asan_support/lib)
 file(GLOB dlls "${O_ASAN_SUPPORT}/../bin/Hostx64/x64/*asan*.dll")
 file(COPY ${dlls} DESTINATION ${CMAKE_BINARY_DIR}/Debug)
 foreach(t o_core original_tests reference_contract candidate_contract owner_controls packet_audit)
  if(TARGET ${t})
   target_include_directories(${t} PRIVATE ${CMAKE_BINARY_DIR}/asan_support)
   target_compile_options(${t} PRIVATE /fsanitize=address /D_DISABLE_VECTOR_ANNOTATION /D_DISABLE_STRING_ANNOTATION /D_DISABLE_OPTIONAL_ANNOTATION)
   target_link_options(${t} PRIVATE /INCREMENTAL:NO /LIBPATH:${CMAKE_BINARY_DIR}/asan_support/lib)
  endif()
 endforeach()
endif()
