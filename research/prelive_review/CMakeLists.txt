cmake_minimum_required(VERSION 3.28)
project(PasPreliveReview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(pixel_review pixel_review.cpp)
target_link_libraries(pixel_review PRIVATE nlohmann_json::nlohmann_json windowscodecs ole32)
if(MSVC)
 target_compile_options(pixel_review PRIVATE /W4 /permissive- /EHsc)
endif()
