cmake_minimum_required(VERSION 3.28)
project(PasTapCostTools LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(PAS_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(nlohmann_json CONFIG REQUIRED)
if(MSVC)
 add_compile_options(/W4 /permissive- /EHsc)
 add_link_options(/STACK:8388608 /MAP)
endif()
add_executable(cost_analysis "${PAS_REPO}/research/prelive_current/cost_analysis.cpp")
add_executable(cost_analysis_tests "${PAS_REPO}/research/prelive_current/tests/cost_analysis_tests.cpp")
add_executable(pas_cost_receipt_verify "${PAS_REPO}/research/windows_cost_receipt/main.cpp")
add_executable(pas_cost_comparison "${PAS_REPO}/research/windows_cost_comparison/main.cpp")
add_executable(comparison_metrics_tests "${PAS_REPO}/research/windows_cost_comparison/metrics_tests.cpp")
foreach(target cost_analysis cost_analysis_tests pas_cost_receipt_verify pas_cost_comparison comparison_metrics_tests)
 target_link_libraries(${target} PRIVATE nlohmann_json::nlohmann_json bcrypt)
endforeach()
