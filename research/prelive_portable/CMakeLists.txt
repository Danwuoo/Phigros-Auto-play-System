cmake_minimum_required(VERSION 3.20)
project(PasPrelivePortableContracts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(PAS_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(PAS_JSON_INCLUDE "" CACHE PATH "Existing pinned patched nlohmann include root; no fetching/installing")
option(PAS_PORTABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(NOT EXISTS "${PAS_JSON_INCLUDE}/nlohmann/json.hpp")
 message(FATAL_ERROR "Supply PAS_JSON_INCLUDE with the existing pinned header")
endif()
file(SHA256 "${PAS_JSON_INCLUDE}/nlohmann/json.hpp" JSON_SHA)
if(NOT JSON_SHA STREQUAL "f2e1f3cac4e48fa851d6986291e4bd5b384d87f92c175d8b126cbd91580289cc")
 message(FATAL_ERROR "JSON source hash differs from the frozen vcpkg-patched dependency")
endif()
if(WIN32 OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 message(FATAL_ERROR "This explicit Linux lane does not replace the Windows prelive_current build")
endif()
find_package(Threads REQUIRED)
add_library(portable_current STATIC
 "${PAS_REPO}/src/core.cpp" "${PAS_REPO}/src/game.cpp" "${PAS_REPO}/src/game_motion.cpp"
 "${PAS_REPO}/src/game_tracking.cpp" "${PAS_REPO}/src/game_session.cpp"
 "${PAS_REPO}/research/bvi_cold_v3/bvi.cpp" "${PAS_REPO}/research/bvi_cold_v3/contact_policy.cpp"
 "${PAS_REPO}/research/bvi_cold_v3/relation_policy.cpp" "${PAS_REPO}/research/bvi_cold_v3/constraint_policy.cpp"
 "${PAS_REPO}/research/bvi_windows/bridge/bridge.cpp" "${PAS_REPO}/research/hold_front_current/front.cpp"
 "${PAS_REPO}/research/prelive_current/current.cpp" "${PAS_REPO}/research/prelive_current/own_bridge.cpp")
target_include_directories(portable_current PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/shim" "${PAS_REPO}/include"
 "${PAS_JSON_INCLUDE}" "${PAS_REPO}/research/bvi_cold_v3" "${PAS_REPO}/research/bvi_windows/bridge"
 "${PAS_REPO}/research/hold_front_current" "${PAS_REPO}/research/prelive_current")
target_compile_options(portable_current PUBLIC -Wall -Wextra -Wpedantic -ffunction-sections -fdata-sections)
target_link_options(portable_current INTERFACE -Wl,--gc-sections)
target_link_libraries(portable_current PUBLIC Threads::Threads)
if(PAS_PORTABLE_SANITIZERS)
 target_compile_options(portable_current PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(portable_current PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
function(add_contract name source)
 add_executable(${name} "${PAS_REPO}/${source}")
 target_link_libraries(${name} PRIVATE portable_current)
 add_test(NAME ${name} COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/run_case.sh" "$<TARGET_FILE:${name}>" "${CMAKE_BINARY_DIR}/reports")
endfunction()
add_contract(original_owner_prefix research/bvi_windows/bridge/bridge_tests.cpp)
add_contract(original_front_contract research/hold_front_current/front_tests.cpp)
add_contract(current_rails_contract research/prelive_current/current_tests.cpp)
add_contract(own_prefix_contract research/prelive_current/own_prefix.cpp)
add_contract(portable_independent research/prelive_portable/independent_tests.cpp)
add_contract(cost_workload_tests research/prelive_current/cost_workload_tests.cpp)
