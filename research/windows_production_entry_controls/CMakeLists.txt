cmake_minimum_required(VERSION 3.28)
project(PasProductionEntryControls LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT PAS_PRODUCTION_SOURCE_REPO OR NOT PAS_PRODUCTION_BUILD_ROOT)
 message(FATAL_ERROR "Frozen exact source and production build required")
endif()
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Protobuf CONFIG REQUIRED)
find_package(gRPC 1.81.1 EXACT CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
foreach(lib pas_emulator pas_current_candidate pas_core pas_proto)
 add_library(${lib} STATIC IMPORTED GLOBAL)
 set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${PAS_PRODUCTION_BUILD_ROOT}/${lib}.lib")
endforeach()
add_executable(production_entry_controls main.cpp)
target_include_directories(production_entry_controls PRIVATE "${PAS_PRODUCTION_SOURCE_REPO}/include" "${PAS_PRODUCTION_BUILD_ROOT}/generated")
target_link_libraries(production_entry_controls PRIVATE pas_emulator pas_current_candidate pas_core pas_proto nlohmann_json::nlohmann_json ZLIB::ZLIB protobuf::libprotobuf gRPC::grpc++ bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32)
if(MSVC)
 target_compile_options(production_entry_controls PRIVATE /W4 /permissive- /EHsc)
endif()
