cmake_minimum_required(VERSION 3.28)
project(PasCurrentSessionSchemaCompatibility LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT PAS_CURRENT_SOURCE OR NOT PAS_CURRENT_CORE)
 message(FATAL_ERROR "Frozen current source/core required")
endif()
find_package(GTest CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
option(PAS_PRELIVE_ASAN "Sanitize compatibility test TU and current imported core" OFF)
if(MSVC)
 add_compile_options(/W4 /permissive- /EHsc)
 add_link_options(/STACK:8388608 /MAP)
 if(PAS_PRELIVE_ASAN)
  set(CMAKE_CXX_FLAGS_DEBUG "/MDd /Od /Ob0 /D_DEBUG")
  add_compile_options(/fsanitize=address)
  add_compile_definitions(_DISABLE_VECTOR_ANNOTATION _DISABLE_STRING_ANNOTATION _DISABLE_OPTIONAL_ANNOTATION)
  file(COPY "${PAS_ASAN_SUPPORT_ROOT}/sanitizer" DESTINATION "${CMAKE_BINARY_DIR}/asan_support")
  file(GLOB ASAN_LIBS "${PAS_ASAN_SUPPORT_ROOT}/../lib/x64/*asan*.lib")
  file(COPY ${ASAN_LIBS} DESTINATION "${CMAKE_BINARY_DIR}/asan_support/lib")
  file(GLOB ASAN_DLLS "${PAS_ASAN_SUPPORT_ROOT}/../bin/Hostx64/x64/*asan*.dll")
  file(COPY ${ASAN_DLLS} DESTINATION "${CMAKE_BINARY_DIR}")
  include_directories("${CMAKE_BINARY_DIR}/asan_support")
  link_directories("${CMAKE_BINARY_DIR}/asan_support/lib")
  add_link_options(/INCREMENTAL:NO)
 endif()
endif()
add_library(current_core STATIC IMPORTED GLOBAL)
set_target_properties(current_core PROPERTIES IMPORTED_LOCATION "${PAS_CURRENT_CORE}")
add_executable(current_schema_compatibility main.cpp)
target_include_directories(current_schema_compatibility PRIVATE "${PAS_CURRENT_SOURCE}/include" "${PAS_CURRENT_SOURCE}/tests")
target_compile_definitions(current_schema_compatibility PRIVATE PAS_TEST_SOURCE_DIR="${PAS_CURRENT_SOURCE}")
target_link_libraries(current_schema_compatibility PRIVATE current_core GTest::gtest_main nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32)
