set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(BVI_ASAN "Instrument isolated candidate and harness" OFF)
add_executable(bvi_tests bvi.cpp main.cpp)
add_executable(bvi_png bvi.cpp png.cpp)
foreach(t bvi_tests bvi_png)
 target_include_directories(${t} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../out/vcpkg_installed/x64-windows/include")
 target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8 /Zi /showIncludes)
 target_link_options(${t} PRIVATE /DEBUG /INCREMENTAL:NO)
 if(BVI_ASAN)
  target_compile_options(${t} PRIVATE /fsanitize=address /D_DISABLE_VECTOR_ANNOTATION /D_DISABLE_STRING_ANNOTATION /D_DISABLE_OPTIONAL_ANNOTATION)
 endif()
endforeach()
target_link_libraries(bvi_png PRIVATE windowscodecs ole32 bcrypt)
