cmake_minimum_required(VERSION 3.28)
project(PasQualifiedProductionBuild LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT PAS_PRODUCTION_SOURCE_REPO)
 message(FATAL_ERROR "Exact frozen source repository required")
endif()
file(READ "${PAS_PRODUCTION_SOURCE_REPO}/CMakeLists.txt" ROOT_SOURCE)
string(REGEX MATCH "add_library\\(pas_core STATIC ([^)]*)\\)" CORE_DECLARATION "${ROOT_SOURCE}")
if(NOT CORE_DECLARATION)
 message(FATAL_ERROR "Original core declaration missing")
endif()
set(CORE_RELATIVE "${CMAKE_MATCH_1}")
separate_arguments(CORE_RELATIVE UNIX_COMMAND "${CORE_RELATIVE}")
set(CORE_SOURCES)
foreach(item IN LISTS CORE_RELATIVE)
 list(APPEND CORE_SOURCES "${PAS_PRODUCTION_SOURCE_REPO}/${item}")
endforeach()
find_package(CLI11 CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Protobuf CONFIG REQUIRED)
find_package(gRPC 1.81.1 EXACT CONFIG REQUIRED)
find_package(FFMPEG REQUIRED)
find_package(ZLIB REQUIRED)
set(PROTO_FILE "${PAS_PRODUCTION_SOURCE_REPO}/proto/emulator_controller.proto")
set(PROTO_OUT "${CMAKE_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${PROTO_OUT}")
add_custom_command(OUTPUT ${PROTO_OUT}/emulator_controller.pb.cc ${PROTO_OUT}/emulator_controller.pb.h ${PROTO_OUT}/emulator_controller.grpc.pb.cc ${PROTO_OUT}/emulator_controller.grpc.pb.h
 COMMAND $<TARGET_FILE:protobuf::protoc> --proto_path=${PAS_PRODUCTION_SOURCE_REPO}/proto --cpp_out=${PROTO_OUT} --grpc_out=${PROTO_OUT} --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> ${PROTO_FILE}
 DEPENDS ${PROTO_FILE} VERBATIM)
add_library(pas_proto STATIC ${PROTO_OUT}/emulator_controller.pb.cc ${PROTO_OUT}/emulator_controller.grpc.pb.cc)
target_include_directories(pas_proto PUBLIC ${PROTO_OUT})
target_link_libraries(pas_proto PUBLIC protobuf::libprotobuf gRPC::grpc++)
add_library(pas_core STATIC ${CORE_SOURCES})
target_include_directories(pas_core PUBLIC "${PAS_PRODUCTION_SOURCE_REPO}/include")
target_link_libraries(pas_core PUBLIC nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32)
string(REGEX MATCH "add_library\\(pas_current_candidate STATIC[ \r\n]+([^)]*)\\)" CURRENT_DECLARATION "${ROOT_SOURCE}")
if(NOT CURRENT_DECLARATION)
 message(FATAL_ERROR "Original current candidate declaration missing")
endif()
set(CURRENT_SOURCES "${CMAKE_MATCH_1}")
separate_arguments(CURRENT_SOURCES UNIX_COMMAND "${CURRENT_SOURCES}")
list(TRANSFORM CURRENT_SOURCES PREPEND "${PAS_PRODUCTION_SOURCE_REPO}/")
add_library(pas_current_candidate STATIC ${CURRENT_SOURCES})
target_include_directories(pas_current_candidate PUBLIC "${PAS_PRODUCTION_SOURCE_REPO}/research/bvi_cold_v3" "${PAS_PRODUCTION_SOURCE_REPO}/research/hold_front_current" "${PAS_PRODUCTION_SOURCE_REPO}/research/prelive_current")
target_link_libraries(pas_current_candidate PUBLIC pas_core)
set(EMULATOR_SOURCES src/emulator.cpp src/grpc_transport.cpp src/runtime.cpp src/manual_session.cpp)
list(TRANSFORM EMULATOR_SOURCES PREPEND "${PAS_PRODUCTION_SOURCE_REPO}/")
add_library(pas_emulator STATIC ${EMULATOR_SOURCES})
target_include_directories(pas_emulator PUBLIC "${PAS_PRODUCTION_SOURCE_REPO}/include" PRIVATE ${PROTO_OUT})
target_link_libraries(pas_emulator PUBLIC pas_core pas_proto pas_current_candidate)
# Use the unchanged root's complete build-provenance block with its exact
# relative hash inventory. No Windows endpoint is executed by this project.
string(FIND "${ROOT_SOURCE}" "execute_process(COMMAND git rev-parse HEAD" PROVENANCE_START)
string(FIND "${ROOT_SOURCE}" "target_include_directories(pas_emulator PRIVATE" PROVENANCE_END)
if(PROVENANCE_START LESS 0 OR PROVENANCE_END LESS PROVENANCE_START)
 message(FATAL_ERROR "Original provenance block missing")
endif()
math(EXPR PROVENANCE_LENGTH "${PROVENANCE_END}-${PROVENANCE_START}")
string(SUBSTRING "${ROOT_SOURCE}" ${PROVENANCE_START} ${PROVENANCE_LENGTH} PROVENANCE_SOURCE)
string(REPLACE "CMAKE_CURRENT_SOURCE_DIR" "PAS_PRODUCTION_SOURCE_REPO" PROVENANCE_SOURCE "${PROVENANCE_SOURCE}")
string(REPLACE "configure_file(cmake/" "configure_file(\"${PAS_PRODUCTION_SOURCE_REPO}/cmake/" PROVENANCE_SOURCE "${PROVENANCE_SOURCE}")
string(REPLACE "session_build_provenance.hpp.in generated/" "session_build_provenance.hpp.in\" generated/" PROVENANCE_SOURCE "${PROVENANCE_SOURCE}")
file(WRITE "${CMAKE_BINARY_DIR}/production-provenance.cmake" "${PROVENANCE_SOURCE}")
include("${CMAKE_BINARY_DIR}/production-provenance.cmake")
set(BENCH_SOURCES src/bench.cpp src/bench_workloads.cpp src/touch_bench.cpp src/native_capture.cpp src/scrcpy_capture.cpp)
list(TRANSFORM BENCH_SOURCES PREPEND "${PAS_PRODUCTION_SOURCE_REPO}/")
add_library(pas_bench STATIC ${BENCH_SOURCES})
target_include_directories(pas_bench PUBLIC "${PAS_PRODUCTION_SOURCE_REPO}/include" PRIVATE ${FFMPEG_INCLUDE_DIRS})
target_link_directories(pas_bench PRIVATE ${FFMPEG_LIBRARY_DIRS})
target_link_libraries(pas_bench PUBLIC pas_emulator pas_core PRIVATE psapi pdh windowsapp runtimeobject ws2_32 dwmapi ${FFMPEG_LIBRARIES})
add_executable(pas "${PAS_PRODUCTION_SOURCE_REPO}/apps/pas/main.cpp")
target_link_libraries(pas PRIVATE pas_core pas_emulator pas_bench CLI11::CLI11)
if(MSVC)
 foreach(target pas_proto pas_core pas_current_candidate pas_emulator pas_bench pas)
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
 endforeach()
endif()
