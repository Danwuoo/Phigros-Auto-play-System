function(pas_original_static_sources target output expected_count)
  string(REGEX MATCHALL
    "add_library\\(${target}[ \t\r\n]+STATIC[ \t\r\n]+[^)]*\\)"
    declarations "${ROOT_SOURCE}")
  list(LENGTH declarations declaration_count)
  if(NOT declaration_count EQUAL 1)
    message(FATAL_ERROR "One original literal ${target} declaration required")
  endif()
  list(GET declarations 0 declaration)
  string(REGEX MATCH
    "add_library\\(${target}[ \t\r\n]+STATIC[ \t\r\n]+([^)]*)\\)"
    matched "${declaration}")
  set(relative_sources "${CMAKE_MATCH_1}")
  separate_arguments(relative_sources UNIX_COMMAND "${relative_sources}")
  list(LENGTH relative_sources source_count)
  set(unique_sources ${relative_sources})
  list(REMOVE_DUPLICATES unique_sources)
  list(LENGTH unique_sources unique_count)
  if(NOT source_count EQUAL expected_count OR NOT unique_count EQUAL expected_count)
    message(FATAL_ERROR "Complete unique original ${target} source set required")
  endif()
  set(absolute_sources)
  foreach(source IN LISTS relative_sources)
    if(NOT source MATCHES "^[A-Za-z0-9_/-]+\\.cpp$" OR
       source MATCHES "(^|/)\\.\\.?(/|$)" OR
       NOT EXISTS "${PAS_PRODUCTION_SOURCE_REPO}/${source}")
      message(FATAL_ERROR "Original literal source missing/unsafe: ${source}")
    endif()
    list(APPEND absolute_sources "${PAS_PRODUCTION_SOURCE_REPO}/${source}")
  endforeach()
  set(${output} "${absolute_sources}" PARENT_SCOPE)
endfunction()
