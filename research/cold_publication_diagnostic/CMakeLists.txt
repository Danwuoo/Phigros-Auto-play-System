cmake_minimum_required(VERSION 3.28)
project(PasColdPublicationDiagnostic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(VCPKG_APPLOCAL_DEPS OFF CACHE BOOL "Explicit frozen runtime copies" FORCE)
foreach(name PAS_SOURCE_WORKSPACE PAS_SOURCE_COMMIT PAS_LIBRARY_INPUTS_FILE PAS_RUNTIME_INPUTS_FILE PAS_BASE_MOTION_OBJECT PAS_BASE_MOTION_SHA PAS_WORKER_AFFINITY_SHA PAS_PRODUCER_PLACEMENT_SHA)
 if(NOT DEFINED ${name} OR "${${name}}" STREQUAL "")
  message(FATAL_ERROR "Explicit diagnostic input required: ${name}")
 endif()
endforeach()
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
include("${PAS_LIBRARY_INPUTS_FILE}")
include("${PAS_RUNTIME_INPUTS_FILE}")
foreach(role cpu_current_rails cpu_current_front cpu_exact_bvi_v3 cpu_own_bridge cpu_pas_core)
 file(SHA256 "${PAS_${role}_PATH}" actual)
 if(NOT actual STREQUAL PAS_${role}_SHA256)
  message(FATAL_ERROR "Original archive SHA: ${role}")
 endif()
 add_library(${role} STATIC IMPORTED GLOBAL)
 set_target_properties(${role} PROPERTIES IMPORTED_LOCATION "${PAS_${role}_PATH}")
endforeach()
file(SHA256 "${PAS_BASE_MOTION_OBJECT}" motion_actual)
if(NOT motion_actual STREQUAL PAS_BASE_MOTION_SHA)
 message(FATAL_ERROR "Exact original main80 motion object required")
endif()
set(DIAG "${PAS_SOURCE_WORKSPACE}/research/cold_publication_diagnostic")
set(RAILS "${PAS_SOURCE_WORKSPACE}/research/prelive_current")
file(WRITE "${CMAKE_BINARY_DIR}/producer_eligibility_delta_identity.hpp"
 "#pragma once\ninline constexpr auto pas_producer_eligibility_delta_source_commit=\"${PAS_SOURCE_COMMIT}\";\ninline constexpr bool pas_producer_eligibility_delta_source_dirty=false;\ninline constexpr auto pas_producer_eligibility_delta_base_source_commit=\"12ec0550e882e3b06751d67fa67c43dda3e9312b\";\ninline constexpr auto pas_producer_eligibility_delta_worker_affinity_sha256=\"${PAS_WORKER_AFFINITY_SHA}\";\ninline constexpr auto pas_producer_eligibility_delta_producer_placement_sha256=\"${PAS_PRODUCER_PLACEMENT_SHA}\";\n")
add_library(publication_core OBJECT "${DIAG}/core.instrumented.cpp")
set_source_files_properties("${PAS_BASE_MOTION_OBJECT}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_executable(current_chain "${DIAG}/chain.instrumented.cpp" $<TARGET_OBJECTS:publication_core> "${PAS_BASE_MOTION_OBJECT}")
target_link_libraries(current_chain PRIVATE cpu_current_rails cpu_current_front cpu_own_bridge cpu_exact_bvi_v3 cpu_pas_core nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32 psapi)
add_executable(publication_controls "${DIAG}/controls.cpp" $<TARGET_OBJECTS:publication_core>)
target_link_libraries(publication_controls PRIVATE nlohmann_json::nlohmann_json bcrypt)
foreach(target publication_core current_chain publication_controls)
 target_include_directories(${target} PRIVATE "${PAS_SOURCE_WORKSPACE}" "${PAS_SOURCE_WORKSPACE}/include" "${DIAG}" "${RAILS}" "${PAS_SOURCE_WORKSPACE}/research/bvi_cold_v3" "${PAS_SOURCE_WORKSPACE}/research/hold_front_current" "${CMAKE_BINARY_DIR}")
 target_compile_definitions(${target} PRIVATE PAS_PUBLICATION_DIAGNOSTIC_OVERLAY=1 PAS_PUBLICATION_DIAGNOSTIC_SOURCE_COMMIT="${PAS_SOURCE_COMMIT}" PAS_PUBLICATION_DIAGNOSTIC_SOURCE_DIRTY=0)
 target_link_libraries(${target} PRIVATE nlohmann_json::nlohmann_json)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
 endif()
endforeach()
target_compile_definitions(current_chain PRIVATE PAS_PRODUCER_ELIGIBILITY_DELTA=1)
if(MSVC)
 foreach(target current_chain publication_controls)
  target_link_options(${target} PRIVATE /STACK:8388608 "/MAP:${CMAKE_BINARY_DIR}/${target}.map" /VERBOSE:LIB)
 endforeach()
endif()
foreach(runtime IN LISTS PAS_RUNTIME_PATHS)
 get_filename_component(name "${runtime}" NAME)
 file(COPY "${runtime}" DESTINATION "${CMAKE_BINARY_DIR}")
endforeach()
