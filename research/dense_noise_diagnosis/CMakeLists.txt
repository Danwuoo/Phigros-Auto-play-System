cmake_minimum_required(VERSION 3.28)
project(PasDenseNoiseDiagnosis LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(dense_noise_diagnosis main.cpp)
target_link_libraries(dense_noise_diagnosis PRIVATE nlohmann_json::nlohmann_json)
if(MSVC)
 target_compile_options(dense_noise_diagnosis PRIVATE /W4 /permissive- /EHsc)
endif()
