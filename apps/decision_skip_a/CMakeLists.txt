cmake_minimum_required(VERSION 3.28)
project(DecisionSkipA LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(decision_skip_a main.cpp)
target_link_libraries(decision_skip_a PRIVATE nlohmann_json::nlohmann_json bcrypt)
target_compile_options(decision_skip_a PRIVATE /W4 /EHsc /utf-8 /bigobj)
target_link_options(decision_skip_a PRIVATE /MAP /INCREMENTAL:NO)
if(SKIP_A_ASAN)
 target_compile_options(decision_skip_a PRIVATE /fsanitize=address /D_DISABLE_VECTOR_ANNOTATION /D_DISABLE_STRING_ANNOTATION /D_DISABLE_OPTIONAL_ANNOTATION)
 get_filename_component(SKIP_A_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
 target_link_options(decision_skip_a PRIVATE "/LIBPATH:${SKIP_A_REPO}/out/x11-p-r1/build-asan/asan_support/lib")
 add_custom_command(TARGET decision_skip_a POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SKIP_A_REPO}/out/x11-p-r1/build-asan/Debug/clang_rt.asan_dynamic-x86_64.dll" "$<TARGET_FILE_DIR:decision_skip_a>")
endif()
