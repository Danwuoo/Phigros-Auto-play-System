cmake_minimum_required(VERSION 3.28)
project(R3BoundedCost LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
find_package(GTest CONFIG REQUIRED)
if(R3_ASAN)
 set(R3_CORE "${R3_REPO}/out/x11-p-r1/build-asan/Debug/pas_core.lib")
else()
 set(R3_CORE "${R3_REPO}/out/x11-p-r1/build-${R3_ROLE}/Release/pas_core.lib")
endif()
add_executable(r3_meter main.cpp session_archive.cpp)
add_executable(r3_tests tests.cpp session_archive.cpp)
add_executable(r3_audit audit.cpp)
add_executable(r3_gate gate.cpp)
foreach(t r3_meter r3_tests r3_audit r3_gate)
 target_include_directories(${t} PRIVATE "${R3_REPO}/out/x11-p-r1/${R3_ROLE}/include")
 target_link_libraries(${t} PRIVATE nlohmann_json::nlohmann_json)
 target_compile_options(${t} PRIVATE /W4 /EHsc /utf-8 /bigobj)
 if(R3_ASAN)
  target_include_directories(${t} BEFORE PRIVATE "${R3_REPO}/out/x11-p-r1/build-asan/asan_support")
  target_compile_options(${t} PRIVATE /fsanitize=address /D_DISABLE_VECTOR_ANNOTATION /D_DISABLE_STRING_ANNOTATION /D_DISABLE_OPTIONAL_ANNOTATION)
  target_link_options(${t} PRIVATE /INCREMENTAL:NO "/LIBPATH:${R3_REPO}/out/x11-p-r1/build-asan/asan_support/lib")
  foreach(d clang_rt.asan_dynamic-x86_64.dll clang_rt.asan_dbg_dynamic-x86_64.dll zd.dll)
   add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${R3_REPO}/out/x11-p-r1/build-asan/Debug/${d}" "$<TARGET_FILE_DIR:${t}>")
  endforeach()
 endif()
endforeach()
foreach(t r3_meter r3_tests)
 target_link_libraries(${t} PRIVATE "${R3_CORE}" ZLIB::ZLIB bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32 psapi)
 target_link_options(${t} PRIVATE /MAP /VERBOSE:LIB)
endforeach()
target_link_libraries(r3_tests PRIVATE GTest::gtest_main)
