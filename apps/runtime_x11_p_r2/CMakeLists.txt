cmake_minimum_required(VERSION 3.28)
project(R2OfflineCoverage LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
find_package(GTest CONFIG REQUIRED)
if(R2_ASAN)
 set(R2_CORE "${R2_REPO}/out/x11-p-r1/build-asan/Debug/pas_core.lib")
else()
 set(R2_CORE "${R2_REPO}/out/x11-p-r1/build-${R2_ROLE}/Release/pas_core.lib")
endif()
add_executable(r2_active_meter main.cpp)
add_executable(r2_tests tests.cpp)
foreach(t r2_active_meter r2_tests)
 target_include_directories(${t} PRIVATE "${R2_REPO}/out/x11-p-r1/${R2_ROLE}/include")
 target_link_libraries(${t} PRIVATE "${R2_CORE}" nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 d3d11 dxgi d3dcompiler user32 psapi)
 target_compile_options(${t} PRIVATE /W4 /EHsc /utf-8 /bigobj)
 if(R2_ASAN)
  target_include_directories(${t} BEFORE PRIVATE "${R2_REPO}/out/x11-p-r1/build-asan/asan_support")
  target_compile_options(${t} PRIVATE /fsanitize=address /D_DISABLE_VECTOR_ANNOTATION /D_DISABLE_STRING_ANNOTATION /D_DISABLE_OPTIONAL_ANNOTATION)
  target_link_options(${t} PRIVATE /INCREMENTAL:NO "/LIBPATH:${R2_REPO}/out/x11-p-r1/build-asan/asan_support/lib")
  foreach(d clang_rt.asan_dynamic-x86_64.dll clang_rt.asan_dbg_dynamic-x86_64.dll zd.dll)
   add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${R2_REPO}/out/x11-p-r1/build-asan/Debug/${d}" "$<TARGET_FILE_DIR:${t}>")
  endforeach()
 endif()
endforeach()
target_link_libraries(r2_tests PRIVATE GTest::gtest_main)
