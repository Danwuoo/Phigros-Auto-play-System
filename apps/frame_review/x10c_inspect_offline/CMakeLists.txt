cmake_minimum_required(VERSION 3.28)
project(PasHoldCascadeInspection LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(x10c_inspect ../x10c_inspect.cpp)
target_link_libraries(x10c_inspect PRIVATE nlohmann_json::nlohmann_json)
target_compile_options(x10c_inspect PRIVATE /W4 /EHsc /utf-8)
add_executable(x10c_causal_report ../x10c_causal_report.cpp)
target_link_libraries(x10c_causal_report PRIVATE nlohmann_json::nlohmann_json)
target_compile_options(x10c_causal_report PRIVATE /W4 /EHsc /utf-8)
if(X10C_CORE AND X10C_INCLUDE)
  find_package(ZLIB REQUIRED)
  add_executable(x10c_pending_probe ../x10c_pending_probe.cpp)
  target_include_directories(x10c_pending_probe PRIVATE ${X10C_INCLUDE})
  target_link_libraries(x10c_pending_probe PRIVATE ${X10C_CORE} nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 user32)
  target_compile_options(x10c_pending_probe PRIVATE /W4 /EHsc /utf-8)
endif()
