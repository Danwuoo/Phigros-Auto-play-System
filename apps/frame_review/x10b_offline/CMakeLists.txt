cmake_minimum_required(VERSION 3.28)
project(PasCurrentInteriorClaim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(GTest CONFIG REQUIRED)
add_subdirectory(../offline reused)
add_executable(x10b_claim_tests ${X1_REPO}/tests/x10b_claim_tests.cpp)
target_link_libraries(x10b_claim_tests PRIVATE x1_core GTest::gtest_main)
target_compile_options(x10b_claim_tests PRIVATE /W4 /EHsc /utf-8)
add_executable(x10b_audit ../x10b_audit.cpp)
target_link_libraries(x10b_audit PRIVATE x1_core)
target_compile_options(x10b_audit PRIVATE /W4 /EHsc /utf-8)
if(X1_SANITIZER_SUPPORT)
  target_include_directories(x10b_claim_tests PRIVATE ${CMAKE_BINARY_DIR}/asan_support)
  target_compile_options(x10b_claim_tests PRIVATE $<$<CONFIG:Debug>:/fsanitize=address> $<$<CONFIG:Debug>:/D_DISABLE_VECTOR_ANNOTATION> $<$<CONFIG:Debug>:/D_DISABLE_STRING_ANNOTATION> $<$<CONFIG:Debug>:/D_DISABLE_OPTIONAL_ANNOTATION>)
  target_link_options(x10b_claim_tests PRIVATE $<$<CONFIG:Debug>:/INCREMENTAL:NO> $<$<CONFIG:Debug>:/LIBPATH:${CMAKE_BINARY_DIR}/asan_support/lib>)
endif()
