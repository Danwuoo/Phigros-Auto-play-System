function(lineage name root)
 set(sources)
 foreach(p IN LISTS parts)
  list(APPEND sources ${root}/src/${p}.cpp)
 endforeach()
 add_library(${name}_core STATIC ${sources})
 target_include_directories(${name}_core PUBLIC ${root}/include ${X1_REPO}/apps/frame_review)
 target_link_libraries(${name}_core PUBLIC nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 user32)
 target_compile_options(${name}_core PRIVATE /W4 /permissive- /EHsc /utf-8)
 add_executable(${name}_replay ${X1_REPO}/apps/frame_review/x10c_main.cpp ${X1_REPO}/apps/frame_review/x10d_p_contact_replay.cpp)
 target_link_libraries(${name}_replay PRIVATE ${name}_core)
 target_compile_options(${name}_replay PRIVATE /W4 /EHsc /utf-8 /bigobj)
 add_executable(${name}_tests ${X1_REPO}/tests/contact_replay_tests.cpp ${X1_REPO}/apps/frame_review/x10d_p_contact_replay.cpp
  ${root}/tests/game_tests_offline.cpp ${root}/tests/game_tracking_tests.cpp ${root}/tests/game_session_tests.cpp ${root}/tests/core_tests.cpp)
 target_include_directories(${name}_tests PRIVATE ${root}/tests)
 target_compile_definitions(${name}_tests PRIVATE PAS_TEST_SOURCE_DIR="${X1_REPO}")
 target_compile_options(${name}_tests PRIVATE /EHsc /utf-8 /bigobj)
 target_link_libraries(${name}_tests PRIVATE ${name}_core GTest::gtest_main)
 add_executable(${name}_pending_tests ${X1_REPO}/tests/x10d_p_owner_tests.cpp)
 target_link_libraries(${name}_pending_tests PRIVATE ${name}_core GTest::gtest_main)
 target_compile_options(${name}_pending_tests PRIVATE /W4 /EHsc /utf-8)
endfunction()
