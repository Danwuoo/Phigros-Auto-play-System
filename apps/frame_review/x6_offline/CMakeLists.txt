cmake_minimum_required(VERSION 3.28)
project(PasPixelReliability LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(GTest CONFIG REQUIRED)
add_subdirectory(../offline reused)
add_executable(x6_report ../x6_report.cpp)
target_link_libraries(x6_report PRIVATE x1_core)
add_executable(x6_tests ${X1_REPO}/tests/x6_pixel_tests.cpp)
target_link_libraries(x6_tests PRIVATE x1_core GTest::gtest_main)
foreach(target x6_report x6_tests)
  target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8 /bigobj)
  if(X1_SANITIZER_SUPPORT)
    target_include_directories(${target} PRIVATE ${CMAKE_BINARY_DIR}/asan_support)
    target_compile_options(${target} PRIVATE $<$<CONFIG:Debug>:/fsanitize=address> $<$<CONFIG:Debug>:/D_DISABLE_VECTOR_ANNOTATION> $<$<CONFIG:Debug>:/D_DISABLE_STRING_ANNOTATION> $<$<CONFIG:Debug>:/D_DISABLE_OPTIONAL_ANNOTATION>)
    target_link_options(${target} PRIVATE $<$<CONFIG:Debug>:/INCREMENTAL:NO> $<$<CONFIG:Debug>:/LIBPATH:${CMAKE_BINARY_DIR}/asan_support/lib>)
  endif()
endforeach()
enable_testing()
add_test(NAME x6_tests COMMAND x6_tests --gtest_brief=1)
