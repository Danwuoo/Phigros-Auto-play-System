cmake_minimum_required(VERSION 3.28)
project(PasContactReplay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT X1_SOURCE OR NOT X1_REPO)
  message(FATAL_ERROR "X1_SOURCE and X1_REPO required; prepare with prepare-contact-replay.ps1")
endif()
file(TO_CMAKE_PATH "${X1_SOURCE}" X1_SOURCE)
file(TO_CMAKE_PATH "${X1_REPO}" X1_REPO)
if(X1_SANITIZER_SUPPORT)
  if(NOT EXISTS "${X1_SANITIZER_SUPPORT}/sanitizer/common_interface_defs.h")
    message(FATAL_ERROR "X1_SANITIZER_SUPPORT lacks sanitizer headers")
  endif()
  file(COPY "${X1_SANITIZER_SUPPORT}/sanitizer" DESTINATION "${CMAKE_BINARY_DIR}/asan_support")
  file(GLOB X1_ASAN_LIBS "${X1_SANITIZER_SUPPORT}/../lib/x64/*asan*.lib")
  file(COPY ${X1_ASAN_LIBS} DESTINATION "${CMAKE_BINARY_DIR}/asan_support/lib")
  file(GLOB X1_ASAN_DLLS "${X1_SANITIZER_SUPPORT}/../bin/Hostx64/x64/*asan*.dll")
  file(COPY ${X1_ASAN_DLLS} DESTINATION "${CMAKE_BINARY_DIR}/Debug")
  include_directories("${CMAKE_BINARY_DIR}/asan_support")
  add_compile_options($<$<CONFIG:Debug>:/fsanitize=address>
    $<$<CONFIG:Debug>:/D_DISABLE_VECTOR_ANNOTATION>
    $<$<CONFIG:Debug>:/D_DISABLE_STRING_ANNOTATION>
    $<$<CONFIG:Debug>:/D_DISABLE_OPTIONAL_ANNOTATION>)
  add_link_options($<$<CONFIG:Debug>:/INCREMENTAL:NO>
    $<$<CONFIG:Debug>:/LIBPATH:${CMAKE_BINARY_DIR}/asan_support/lib>)
endif()
find_package(nlohmann_json CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
find_package(GTest CONFIG REQUIRED)
add_library(x1_core STATIC
  ${X1_SOURCE}/src/core.cpp ${X1_SOURCE}/src/game.cpp
  ${X1_SOURCE}/src/game_tracking.cpp ${X1_SOURCE}/src/game_motion.cpp
  ${X1_SOURCE}/src/game_session.cpp ${X1_SOURCE}/src/analysis.cpp
  ${X1_SOURCE}/src/adb.cpp ${X1_SOURCE}/src/config.cpp)
target_sources(x1_core PRIVATE ${X1_SOURCE}/src/game_tracking_methods.cpp
  ${X1_SOURCE}/src/touch_evidence.cpp ${X1_SOURCE}/src/session_archive.cpp
  ${X1_SOURCE}/src/session_pixel_clips.cpp ${X1_SOURCE}/src/journal.cpp
  ${X1_SOURCE}/src/game_tracking_shadow.cpp)
target_include_directories(x1_core PUBLIC ${X1_SOURCE}/include ${X1_REPO}/apps/frame_review)
target_link_libraries(x1_core PUBLIC nlohmann_json::nlohmann_json ZLIB::ZLIB bcrypt windowscodecs ole32 user32)
target_compile_options(x1_core PRIVATE /W4 /permissive- /EHsc /utf-8)
add_executable(pas_frame_review ${X1_REPO}/apps/frame_review/main.cpp ${X1_REPO}/apps/frame_review/contact_replay.cpp)
target_link_libraries(pas_frame_review PRIVATE x1_core)
target_compile_definitions(pas_frame_review PRIVATE PAS_X1_OFFLINE=1)
target_compile_options(pas_frame_review PRIVATE /W4 /EHsc /utf-8)
add_executable(x1_subject_prefix_report ${X1_REPO}/apps/frame_review/subject_prefix_report.cpp)
target_link_libraries(x1_subject_prefix_report PRIVATE x1_core)
target_compile_options(x1_subject_prefix_report PRIVATE /W4 /EHsc /utf-8)
# Preparation retains all gameplay cases and records two excluded transport
# tests. Their runtime dependency is intentionally absent from this tool.
add_executable(x1_tests ${X1_REPO}/tests/contact_replay_tests.cpp
  ${X1_REPO}/apps/frame_review/contact_replay.cpp
  ${X1_SOURCE}/tests/game_tests_offline.cpp ${X1_SOURCE}/tests/game_tracking_tests.cpp
  ${X1_SOURCE}/tests/game_session_tests.cpp ${X1_SOURCE}/tests/core_tests.cpp)
target_link_libraries(x1_tests PRIVATE x1_core GTest::gtest_main)
target_compile_definitions(x1_tests PRIVATE PAS_TEST_SOURCE_DIR="${X1_REPO}")
target_compile_options(x1_tests PRIVATE /EHsc /utf-8 /bigobj)
target_include_directories(x1_tests PRIVATE ${X1_SOURCE}/tests)
if(X2_OFFLINE)
  add_executable(x2_report ${X1_REPO}/apps/frame_review/x2_report.cpp)
  target_link_libraries(x2_report PRIVATE x1_core)
  target_compile_options(x2_report PRIVATE /W4 /EHsc /utf-8)
  target_compile_definitions(pas_frame_review PRIVATE PAS_X2_OFFLINE=1)
  target_compile_definitions(x1_tests PRIVATE PAS_X2_OFFLINE=1)
  target_sources(x1_tests PRIVATE ${X1_REPO}/tests/x2_ablation_tests.cpp)
  if(X2_MAIN50)
    target_compile_definitions(pas_frame_review PRIVATE PAS_X2_MAIN50=1)
    target_compile_definitions(x1_tests PRIVATE PAS_X2_MAIN50=1)
  endif()
endif()
if(X3_REPORT)
  add_executable(x3_report ${X1_REPO}/apps/frame_review/x3_main.cpp ${X1_REPO}/apps/frame_review/x3_report.cpp)
  target_link_libraries(x3_report PRIVATE x1_core)
  target_compile_options(x3_report PRIVATE /W4 /EHsc /utf-8 /bigobj)
  add_executable(x3_tests ${X1_REPO}/tests/x3_features_tests.cpp ${X1_REPO}/apps/frame_review/x3_report.cpp)
  target_link_libraries(x3_tests PRIVATE x1_core GTest::gtest_main)
  target_compile_options(x3_tests PRIVATE /W4 /EHsc /utf-8 /bigobj)
endif()
if(X4_OFFLINE)
  if(X2_OFFLINE)
    message(FATAL_ERROR "X4 must not combine with the X2 ablation export")
  endif()
  target_compile_definitions(pas_frame_review PRIVATE PAS_X4_OFFLINE=1)
  target_compile_definitions(x1_tests PRIVATE PAS_X4_OFFLINE=1)
  target_sources(x1_tests PRIVATE ${X1_REPO}/tests/x4_oracle_tests.cpp)
  add_executable(x4_tool ${X1_REPO}/apps/frame_review/x4_report.cpp)
  target_link_libraries(x4_tool PRIVATE x1_core)
  target_compile_options(x4_tool PRIVATE /W4 /EHsc /utf-8 /bigobj)
endif()
enable_testing()
add_test(NAME x1_tests COMMAND x1_tests --gtest_brief=1)
