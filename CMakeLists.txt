cmake_minimum_required(VERSION 3.20)
project(RhythmChartMemory VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(rhythm_memory INTERFACE)
target_include_directories(rhythm_memory INTERFACE include)
add_executable(rhythm_dry_run apps/chart_memory/main.cpp)
target_link_libraries(rhythm_dry_run PRIVATE rhythm_memory)
include(CTest)
if(BUILD_TESTING)
  add_executable(rhythm_memory_tests tests/chart_memory_tests.cpp)
  target_link_libraries(rhythm_memory_tests PRIVATE rhythm_memory)
  add_test(NAME rhythm_memory_contracts COMMAND rhythm_memory_tests)
  add_test(NAME phigros_synthetic_dry_run COMMAND rhythm_dry_run phigros)
  add_test(NAME sekai_synthetic_dry_run COMMAND rhythm_dry_run project-sekai)
endif()

option(RHYTHM_BUILD_PAS_CORE "Build the unchanged Windows PAS core and ADB helper" OFF)
option(RHYTHM_BUILD_PAS_GRPC "Build the unchanged patched-gRPC emulator backend" OFF)
option(RHYTHM_BUILD_PAS_TESTS "Build retained PAS core GTest tests" OFF)
if(RHYTHM_BUILD_PAS_CORE OR RHYTHM_BUILD_PAS_GRPC OR RHYTHM_BUILD_PAS_TESTS)
  if(NOT WIN32)
    message(FATAL_ERROR "Retained PAS sources require Windows; portable memory targets do not")
  endif()
  add_library(pas_core STATIC src/core.cpp src/touch_evidence.cpp)
  target_include_directories(pas_core PUBLIC include)
  add_library(pas_adb STATIC src/adb.cpp)
  target_include_directories(pas_adb PUBLIC include)
  target_link_libraries(pas_adb PUBLIC pas_core windowscodecs ole32)
endif()
if(RHYTHM_BUILD_PAS_GRPC)
  include(cmake/PasGrpc.cmake)
endif()
if(RHYTHM_BUILD_PAS_TESTS)
  find_package(GTest CONFIG REQUIRED)
  add_executable(pas_core_tests tests/core_tests.cpp)
  target_link_libraries(pas_core_tests PRIVATE pas_core GTest::gtest_main)
  if(BUILD_TESTING)
    add_test(NAME retained_pas_core COMMAND pas_core_tests)
  endif()
endif()
